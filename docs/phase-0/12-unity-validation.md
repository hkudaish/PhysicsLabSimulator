```powershell
$unityProject = (Resolve-Path .\src\UnityClient).Path
$unityLog = Join-Path (Get-Location) 'artifacts\unity-phase0-build.log'
New-Item -ItemType Directory -Force artifacts | Out-Null
& $unityEditor -batchmode -quit -projectPath $unityProject -buildTarget StandaloneWindows64 -executeMethod VirtualStem.EditorTools.Phase0Build.BuildWindows -logFile $unityLog
if ($LASTEXITCODE -ne 0) { throw 'Unity scaffold build failed; inspect the log.' }
```

| ID | Input / scenario | Required observation |
